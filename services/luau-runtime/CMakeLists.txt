cmake_minimum_required(VERSION 3.20)
project(LeaderSkills LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 17)
set(LUAU_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(runtime/luau EXCLUDE_FROM_ALL)
add_executable(leader-skill-service main.cpp)
target_include_directories(leader-skill-service PRIVATE runtime/include)
target_link_libraries(leader-skill-service PRIVATE Luau.Compiler Luau.VM)
if(MINGW)
  target_link_options(leader-skill-service PRIVATE -static-libgcc -static-libstdc++ -static)
endif()
